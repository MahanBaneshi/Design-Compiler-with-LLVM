Program -> DeclOrStmtList EOF ;
DeclOrStmtList -> (DeclOrStmt)* ;
DeclOrStmt -> VarDecl | TypedVarDecl | ArrayDecl | Stmt ;

VarDecl -> 'var' VarDeclList ';' ;
VarDeclList -> VarDeclItem (',' VarDeclItem)* ;
VarDeclItem -> IDENTIFIER Type ('=' Expr)? ;

TypedVarDecl -> Type TypedVarDeclList ';' ;
TypedVarDeclList -> TypedVarDeclItem (',' TypedVarDeclItem)* ;
TypedVarDeclItem -> IDENTIFIER ('=' Expr)? ;

Type -> 'int' | 'float' | 'bool' ;

ArrayDecl -> 'array' IDENTIFIER ('=' ArrayInit)? ';' ;
ArrayInit -> ArrayLiteral | ArrayComp ;

ArrayLiteral -> '[' (ExprList)? ']' ;
ExprList -> Expr (',' Expr)* ;
ArrayComp -> '[' Expr 'for' IDENTIFIER 'in' IDENTIFIER ('if' CondExpr)? ']' ;

Stmt -> AssignOpStmt ';' | PrintStmt ';' | MatchStmt | IfStmt | WhileStmt | ForStmt | ForeachStmt | Block ;

AssignOpStmt -> 'ADD' LValue Expr Expr | 'SUB' LValue Expr Expr | 'MUL' LValue Expr Expr | 'DIV' LValue Expr Expr | 'MOD' LValue Expr Expr
             | 'PLE' LValue Expr | 'MIE' LValue Expr | 'AND' LValue BoolAtom BoolAtom | 'OR' LValue BoolAtom BoolAtom | 'INC' LValue | 'DEC' LValue ;

LValue -> IDENTIFIER | IDENTIFIER '[' Expr ']' ;

PrintStmt -> 'print' '(' Expr ')' ;

MatchStmt -> 'match' Expr '{' MatchCaseListOpt '}' ;
MatchCaseListOpt -> ε | MatchCaseList ;
MatchCaseList -> MatchCase (',' MatchCase)* (',')? ;
MatchCase -> MatchPattern '->' MatchBody ;
MatchBody -> PrintStmt | Block | AssignOpStmt ';' ;
MatchPattern -> NUMBER | 'true' | 'false' | IDENTIFIER | '_' ;

IfStmt -> 'if' '(' CondExpr ')' Block (ElseIfPart)* (ElsePart)? ;
ElseIfPart -> 'else' 'if' '(' CondExpr ')' Block ;
ElsePart -> 'else' Block ;

WhileStmt -> 'while' '(' CondExpr ')' Block ;

ForStmt -> 'for' '(' ForInit ';' CondExpr ';' ForUpdate ')' Block ;
ForInit -> ('int' IDENTIFIER '=' Expr) | (IDENTIFIER '=' Expr) | ε ;
ForUpdate -> IDENTIFIER '++' | IDENTIFIER '--' | 'INC' LValue | 'DEC' LValue | 'PLE' LValue Expr | 'MIE' LValue Expr | ε ;

ForeachStmt -> 'foreach' '(' IDENTIFIER 'in' IDENTIFIER ')' Block ;

Block -> '{' (Stmt)* '}' ;

CondExpr -> OrExpr ;
OrExpr -> AndExpr ( '||' AndExpr )* ;
AndExpr -> RelExpr ( '&&' RelExpr )* ;
RelExpr -> Expr RelOp Expr | BoolAtom ;
RelOp -> '==' | '!=' | '>' | '>=' | '<' | '<=' ;
BoolAtom -> 'true' | 'false' | IDENTIFIER | '(' CondExpr ')' ;

Expr -> AddExpr ;
AddExpr -> MulExpr (('+'|'-') MulExpr)* ;
MulExpr -> PowExpr (('*'|'/'|'%') PowExpr)* ;
PowExpr -> UnaryExpr ( '^' UnaryExpr )* ;
UnaryExpr -> ('+'|'-') UnaryExpr | Primary ;

Primary -> NUMBER | IDENTIFIER | IDENTIFIER '[' Expr ']' | CallExpr | '(' Expr ')' ;

CallExpr -> ('to_int'|'to_float'|'to_bool'|'abs'|'length'|'max') '(' ArgListOpt ')'
          | 'index' '(' Expr ',' Expr ')'
          | 'find' '(' Expr ',' CondExpr ')' ;

ArgListOpt -> ε | ArgList ;
ArgList -> Expr (',' Expr)* ;
